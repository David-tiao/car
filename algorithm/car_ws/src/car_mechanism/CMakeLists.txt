cmake_minimum_required(VERSION 3.8)
project(car_mechanism)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(car_interfaces REQUIRED)
add_executable(car_mechanism_monitor src/mechanism_monitor_node.cpp)
target_compile_features(car_mechanism_monitor PRIVATE cxx_std_17)
ament_target_dependencies(car_mechanism_monitor rclcpp car_interfaces)
install(TARGETS car_mechanism_monitor DESTINATION lib/${PROJECT_NAME})
ament_package()
