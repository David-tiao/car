cmake_minimum_required(VERSION 3.8)
project(car_mission)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(car_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(car_mission_monitor src/mission_monitor_node.cpp)
target_compile_features(car_mission_monitor PRIVATE cxx_std_17)
target_include_directories(car_mission_monitor PRIVATE include)
ament_target_dependencies(car_mission_monitor rclcpp car_interfaces nav_msgs std_msgs)
install(TARGETS car_mission_monitor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_mission_core tests/test_core.cpp)
  target_compile_features(test_mission_core PRIVATE cxx_std_17)
  target_include_directories(test_mission_core PRIVATE include)
  add_test(NAME car_mission_core COMMAND test_mission_core)
endif()
ament_package()
